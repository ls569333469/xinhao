"""B 同期门禁：0 ≤ t − s ≤ max_lag_ms；s>t 或不符 → 不通过。"""
from __future__ import annotations

from typing import Any, Optional


def contemporaneous_ok(t_ms: int, s_ms: int | None, *, max_lag_ms: int = 300_000) -> str:
    """返回 pass / fail / unknown。"""
    if s_ms is None:
        return "unknown"
    try:
        t = int(t_ms)
        s = int(s_ms)
    except (TypeError, ValueError):
        return "unknown"
    delta = t - s
    if delta < 0:
        return "fail"  # s after t
    if delta > int(max_lag_ms):
        return "fail"
    return "pass"


def pick_sample_at_or_before(t_ms: int, samples: list[dict], *, time_key: str = "timestamp", max_lag_ms: int = 300_000) -> tuple[Optional[dict], str, Optional[int]]:
    """选不晚于 t、最接近 t 且满足门禁的样本。返回 (sample, quality, delta_ms)。"""
    if not samples:
        return None, "unknown", None
    best = None
    best_delta = None
    for row in samples:
        if not isinstance(row, dict):
            continue
        s = row.get(time_key)
        if s is None:
            continue
        try:
            s_i = int(s)
        except (TypeError, ValueError):
            continue
        delta = int(t_ms) - s_i
        if delta < 0:
            continue
        if best is None or delta < best_delta:
            best, best_delta = row, delta
    if best is None:
        return None, "unknown", None
    q = contemporaneous_ok(t_ms, int(best[time_key]), max_lag_ms=max_lag_ms)
    return best, q, best_delta
'''

PROC = r'''# -*- coding: utf-8 -*-
"""按 Codex 13.2：先确认 expected h+1，再判断新 h。纯函数，不发网络。"""
from __future__ import annotations

from typing import Any

from b_accumulation_rules import BAR_MS, BASELINE_BARS, evaluate_h_event, evaluate_h1_confirm
from b_accumulation_store import BAccumulationStore, stable_event_id


def process_bars_in_order(
    root,
    *,
    market: str,
    symbol: str,
    bars_asc: list[dict],
    observation_started_open: int | None = None,
    bg4h_for=None,
) -> dict:
    """bars_asc 已去重升序。bg4h_for(open_time)->dict|None。"""
    store = BAccumulationStore(root)
    store.ensure_observation_start()
    confirmed = 0
    new_events = 0
    # pending confirmations: map expected_open -> h_open
    pending = {}
    # seed from existing store events that await confirm
    for rec in store.list_events():
        if rec.get("market") != market or rec.get("symbol") != symbol:
            continue
        exp = rec.get("expected_confirmation_open_time")
        if exp and rec.get("technical_confirmation") in (None, "pending", "unknown"):
            pending[int(exp)] = int(rec["trigger_open_time"])

    for i, b in enumerate(bars_asc):
        ot = int(b["open_time"])
        # 1) if this bar is expected confirmation
        if ot in pending:
            h_open = pending[ot]
            bg = bg4h_for(ot) if callable(bg4h_for) else None
            conf = evaluate_h1_confirm(bars_asc, h_open, bg4h_h1=bg, h1_closed=True)
            store.upsert_event(
                market=market,
                symbol=symbol,
                trigger_open_time=h_open,
                payload={
                    "technical_confirmation": conf.get("technical_confirmation"),
                    "confirm_ratio": conf.get("ratio"),
                    "pending_confirm": conf.get("pending"),
                    "input_fingerprint": f"confirm:{h_open}:{conf.get('technical_confirmation')}:{conf.get('ratio')}",
                },
            )
            confirmed += 1
            pending.pop(ot, None)
        # 2) evaluate as potential new h (need warmup)
        if i < BASELINE_BARS:
            continue
        if observation_started_open is not None and ot < int(observation_started_open):
            continue  # 历史，不新建提醒事件；仍可评估但不计 new_events 提醒
        bg = bg4h_for(ot) if callable(bg4h_for) else None
        ev = evaluate_h_event(bars_asc, i, bg4h=bg)
        if ev.get("technical_event") == "pass":
            exp = ev.get("expected_confirmation_open_time")
            store.upsert_event(
                market=market,
                symbol=symbol,
                trigger_open_time=ot,
                payload={
                    "technical_event": "pass",
                    "event_ratio": ev.get("ratio"),
                    "abs_return_ok": ev.get("abs_return_ok"),
                    "expected_confirmation_open_time": exp,
                    "technical_confirmation": "pending",
                    "pending_confirm": True,
                    "input_fingerprint": f"event:{ot}:{ev.get('ratio')}",
                },
            )
            if exp:
                pending[int(exp)] = ot
            new_events += 1
    return {"market": market, "symbol": symbol, "confirmed": confirmed, "new_events": new_events, "pending": len(pending)}
'''
(ROOT / "b_process_order.py").write_text(PROC, encoding="utf-8", newline="\n")

from __future__ import annotations

import tempfile

class B07Contemporaneous(unittest.TestCase):
    def test_gate_edges(self):
        t = 1_000_000
        self.assertEqual(contemporaneous_ok(t, t), "pass")
        self.assertEqual(contemporaneous_ok(t, t - 300_000), "pass")
        self.assertEqual(contemporaneous_ok(t, t - 300_001), "fail")
        self.assertEqual(contemporaneous_ok(t, t + 1), "fail")
        self.assertEqual(contemporaneous_ok(t, None), "unknown")

    def test_pick_nearest(self):
        t = 1_000_000
        samples = [
            {"timestamp": t - 400_000, "v": 1},
            {"timestamp": t - 100_000, "v": 2},
            {"timestamp": t + 10, "v": 3},
        ]
        samp, q, delta = pick_sample_at_or_before(t, samples)
        self.assertEqual(samp["v"], 2)
        self.assertEqual(q, "pass")
        self.assertEqual(delta, 100_000)
