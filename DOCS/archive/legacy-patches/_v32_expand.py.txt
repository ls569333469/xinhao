def http_json(url: str, params: dict | None = None):
    if params:
        from urllib.parse import urlencode
        url = url + "?" + urlencode(params)
    with urllib.request.urlopen(url, timeout=30) as r:
        return json.loads(r.read().decode("utf-8"))


def build_universe(limit: int = 40) -> list[dict]:
    """USDⓈ-M USDT 永续，按 24h 成交额取前 N（真实目录，不依赖 A 池）。"""
    info = http_json("https://fapi.binance.com/fapi/v1/exchangeInfo")
    tradable = {
        s["symbol"]
        for s in info.get("symbols", [])
        if s.get("contractType") == "PERPETUAL"
        and s.get("quoteAsset") == "USDT"
        and s.get("status") == "TRADING"
    }
    tickers = http_json("https://fapi.binance.com/fapi/v1/ticker/24hr")
    rows = []
    for t in tickers:
        sym = t.get("symbol")
        if sym not in tradable:
            continue
        try:
            qv = float(t.get("quoteVolume") or 0)
        except (TypeError, ValueError):
            qv = 0.0
        rows.append((qv, sym))
    rows.sort(reverse=True)
    out = [{"market": "USDⓈ-M", "symbol": sym} for _, sym in rows[:limit]]
    path = ROOT / "runtime" / "b-observe-universe.json"
    path.write_text(json.dumps({"symbols": out, "source": "binance-fapi-24hr-top", "limit": limit}, ensure_ascii=False, indent=2), encoding="utf-8", newline="\n")
    print("universe", len(out), "top", [x["symbol"] for x in out[:8]])
    return out
