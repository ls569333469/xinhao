"""Binance 1H K线与同期 OI/LSR 拉取。仅在观察开启的任务里调用。"""
from __future__ import annotations

from typing import Any, Callable, Optional

# hosts
USDT_HOST = "https://fapi.binance.com"
COIN_HOST = "https://dapi.binance.com"


def _hosts(market: str) -> tuple[str, str]:
    if market in ("COIN-M", "COIN_M"):
        return COIN_HOST, "/dapi/v1/"
    return USDT_HOST, "/fapi/v1/"


def fetch_1h_klines(
    market: str,
    symbol: str,
    *,
    limit: int = 500,
    requester: Optional[Callable] = None,
) -> list:
    """返回原始 kline 列表；requester(url, params)->(data, error) 与 scanner.safe 同形。"""
    if requester is None:
        from scanner import safe as requester
    host, version = _hosts(market)
    data, err = requester(host + version + "klines", {"symbol": symbol, "interval": "1h", "limit": int(limit)})
    if err:
        raise RuntimeError(f"1h klines: {err}")
    if not isinstance(data, list):
        raise RuntimeError("1h klines: unexpected payload")
    return data


def fetch_oi_lsr_1h(
    market: str,
    symbol: str,
    *,
    pair: str | None = None,
    limit: int = 2,
    requester: Optional[Callable] = None,
) -> dict:
    """同期 1h OI 历史 + 全体账户多空比。缺失则 quality=unknown。"""
    if requester is None:
        from scanner import safe as requester
    host, version = _hosts(market)
    inverse = host == COIN_HOST
    out: dict[str, Any] = {
        "oi_hist": None,
        "lsr": None,
        "oi_quality": "unknown",
        "lsr_quality": "unknown",
        "errors": {},
    }
    # openInterestHist period=1h
    oi_params = {"symbol": symbol, "period": "1h", "limit": limit}
    if inverse:
        # COIN-M uses pair for some data endpoints
        oi_params = {"pair": pair or symbol.replace("USD", ""), "period": "1h", "limit": limit}
    # Binance futures data endpoints
    oi_url = host + "/futures/data/openInterestHist"
    oi, oi_err = requester(oi_url, oi_params)
    if oi_err:
        out["errors"]["oi_hist"] = oi_err
        out["oi_quality"] = "unknown"
    elif isinstance(oi, list) and oi:
        out["oi_hist"] = oi[-1]
        out["oi_quality"] = "pass"
        try:
            out["sumOpenInterestValue"] = float(oi[-1].get("sumOpenInterestValue"))
        except (TypeError, ValueError):
            out["oi_quality"] = "unknown"
    else:
        out["oi_quality"] = "unknown"

    ratio_args = {"pair": pair or symbol, "period": "1h", "limit": limit} if inverse else {"symbol": symbol, "period": "1h", "limit": limit}
    ratio, ratio_err = requester(host + "/futures/data/globalLongShortAccountRatio", ratio_args)
    if ratio_err:
        out["errors"]["lsr"] = ratio_err
        out["lsr_quality"] = "unknown"
    elif isinstance(ratio, list) and ratio:
        out["lsr"] = ratio[-1]
        try:
            out["longShortRatio"] = float(ratio[-1].get("longShortRatio"))
            out["lsr_quality"] = "pass"
        except (TypeError, ValueError):
            out["lsr_quality"] = "unknown"
    else:
        out["lsr_quality"] = "unknown"
    return out


def make_fetch_1h(requester: Optional[Callable] = None) -> Callable[[str, str], list]:
    def _fetch(market: str, symbol: str) -> list:
        return fetch_1h_klines(market, symbol, requester=requester)
    return _fetch
'''

UNI = r'''# -*- coding: utf-8 -*-
"""B 观察标的清单：显式文件优先；否则从 A 候选池只读摘 symbol（不入 A 池逻辑）。"""
from __future__ import annotations

import json
from pathlib import Path


def load_universe(root: Path) -> list[dict]:
    root = Path(root)
    explicit = root / "runtime" / "b-observe-universe.json"
    if explicit.exists():
        raw = json.loads(explicit.read_text(encoding="utf-8"))
        if isinstance(raw, list):
            return [x for x in raw if isinstance(x, dict) and x.get("symbol") and x.get("market")]
        if isinstance(raw, dict) and isinstance(raw.get("symbols"), list):
            return [x for x in raw["symbols"] if isinstance(x, dict) and x.get("symbol") and x.get("market")]
    # fallback: ema candidates file — read-only borrow of symbols, not A filter
    cand = root / "runtime" / "ema-squeeze-candidates.json"
    if cand.exists():
        data = json.loads(cand.read_text(encoding="utf-8"))
        rows = data.get("candidates") or data.get("rows") or data.get("items") or []
        out = []
        seen = set()
        for r in rows:
            if not isinstance(r, dict):
                continue
            m, s = r.get("market"), r.get("symbol")
            if not m or not s or (m, s) in seen:
                continue
            seen.add((m, s))
            out.append({"market": m, "symbol": s})
        return out
    return []
'''

            if "globalLongShortAccountRatio" in url:
