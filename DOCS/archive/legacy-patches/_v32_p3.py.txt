def bars_from_cache(cache: MarketCandleCache, *, market: str, symbol: str, interval: str = "1h") -> list[dict]:
    rows = cache.get_bars(market=market, symbol=symbol, interval=interval)
    out = []
    for r in rows:
        payload = r.get("payload") or {}
        bar = payload.get("bar") or {}
        ot = int(payload.get("open_time") or r.get("open_time"))
        ct = payload.get("close_time")
        if ct is None:
            ct = r.get("close_time")
        item = {
            "open_time": ot,
            "close_time": ct,
            "o": bar.get("o"),
            "h": bar.get("h"),
            "l": bar.get("l"),
            "c": bar.get("c"),
            "q": bar.get("q"),
        }
        out.append(item)
    out.sort(key=lambda b: b["open_time"])
    return out


def _fingerprint(obj: dict) -> str:
    blob = json.dumps(obj, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode("utf-8")
    return hashlib.sha256(blob).hexdigest()


def evaluate_symbol_from_cache(
    root: Path,
    *,
    market: str,
    symbol: str,
    target_close_boundary: int | None = None,
    bg4h: dict | None = None,
    bg4h_h1: dict | None = None,
) -> dict:
    cache = MarketCandleCache(root)
    store = BAccumulationStore(root)
    store.ensure_observation_start()
    bars = bars_from_cache(cache, market=market, symbol=symbol)
    if target_close_boundary is not None:
        # 事件根 h 的 close boundary = open + BAR_MS；只评估边界等于 target 的那根
        h_open = int(target_close_boundary) - BAR_MS
        bars_use = [b for b in bars if b["open_time"] <= h_open + BAR_MS]  # include h+1 if present
    else:
        if not bars:
            return {"market": market, "symbol": symbol, "evaluated": False, "reason": "no_bars"}
        h_open = bars[-1]["open_time"]
        bars_use = bars

    try:
        h_index = next(i for i, b in enumerate(bars_use) if int(b["open_time"]) == int(h_open))
    except StopIteration:
        return {"market": market, "symbol": symbol, "evaluated": False, "reason": "h_not_in_cache", "h_open": h_open}

    if h_index < BASELINE_BARS:
        return {"market": market, "symbol": symbol, "evaluated": False, "reason": "warmup_short", "h_open": h_open}

    ev = evaluate_h_event(bars_use, h_index, bg4h=bg4h)
    h1_open = int(h_open) + BAR_MS
    h1_present = any(int(b["open_time"]) == h1_open for b in bars_use)
    conf = evaluate_h1_confirm(bars_use, h_open, bg4h_h1=bg4h_h1, h1_closed=h1_present)

    fp_src = {
        "h_open": h_open,
        "technical_event": ev.get("technical_event"),
        "ratio": ev.get("ratio"),
        "abs_return_ok": ev.get("abs_return_ok"),
        "confirm": conf.get("technical_confirmation"),
        "confirm_ratio": conf.get("ratio"),
        "bars_n": len(bars_use),
    }
    payload = {
        "technical_event": ev.get("technical_event"),
        "technical_confirmation": conf.get("technical_confirmation"),
        "pending_confirm": conf.get("pending"),
        "event_ratio": ev.get("ratio"),
        "confirm_ratio": conf.get("ratio"),
        "abs_return_ok": ev.get("abs_return_ok"),
        "reasons": list(ev.get("reasons") or []) + list(conf.get("reasons") or []),
        "input_fingerprint": _fingerprint(fp_src),
        "expected_confirmation_open_time": ev.get("expected_confirmation_open_time"),
        "target_close_boundary": target_close_boundary,
    }
    rec = store.upsert_event(market=market, symbol=symbol, trigger_open_time=h_open, payload=payload)
    return {
        "market": market,
        "symbol": symbol,
        "evaluated": True,
        "event_id": rec.get("event_id"),
        "technical_event": rec.get("technical_event"),
        "technical_confirmation": rec.get("technical_confirmation"),
        "event_revision": rec.get("event_revision"),
    }
'''

TEST = r'''# -*- coding: utf-8 -*-
from __future__ import annotations

import tempfile
import unittest
from pathlib import Path

from b_1h_ingest import ingest_symbol_1h
from b_observe_evaluate import evaluate_symbol_from_cache
from market_candle_cache import MarketCandleCache


def _bar(open_t, q, o=100.0, c=100.5):
    # closed 1h
    return [open_t, str(o), str(o + 1), str(o - 1), str(c), "1", open_t + 3_599_999, str(q)]


class P3EvaluateStore(unittest.TestCase):
    def test_warmup_and_event_write(self):
        with tempfile.TemporaryDirectory() as td:
            root = Path(td)
            cache = MarketCandleCache(root)
            base = 1_700_000_000_000
            raw = []
            # 170 bars: 168 baseline + h + h1
            for i in range(170):
                q = 10.0 if i < 168 else (80.0 if i == 168 else 25.0)
                # keep abs return < 3%
                raw.append(_bar(base + i * 3_600_000, q, o=100.0, c=101.0))
            ingest_symbol_1h(cache, market="USDⓈ-M", symbol="BBBUSDT", raw_klines=raw)
            h_open = base + 168 * 3_600_000
            boundary = h_open + 3_600_000
            # bg4h pass ema7>ema25
            out = evaluate_symbol_from_cache(
                root,
                market="USDⓈ-M",
                symbol="BBBUSDT",
                target_close_boundary=boundary,
                bg4h={"ema7": 2.0, "ema25": 1.0},
                bg4h_h1={"ema7": 2.0, "ema25": 1.0},
            )
            self.assertTrue(out["evaluated"])
            self.assertEqual(out["technical_event"], "pass")
            store_path = root / "runtime" / "b-accumulation-events.json"
            self.assertTrue(store_path.exists())


if __name__ == "__main__":
    unittest.main()
'''
