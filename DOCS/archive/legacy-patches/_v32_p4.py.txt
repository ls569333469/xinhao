API = r'''# -*- coding: utf-8 -*-
"""B 观察只读列表与 CSV。与 A 候选池隔离。"""
from __future__ import annotations

import csv
import io
from pathlib import Path

from b_accumulation_store import BAccumulationStore

CSV_COLS = [
    "event_id",
    "market",
    "symbol",
    "trigger_open_time",
    "technical_event",
    "technical_confirmation",
    "pending_confirm",
    "event_ratio",
    "confirm_ratio",
    "abs_return_ok",
    "event_revision",
    "first_seen_at",
    "last_evaluated_at",
    "expected_confirmation_open_time",
]


def events_payload(root: Path) -> dict:
    store = BAccumulationStore(root)
    meta = store.meta()
    events = store.list_events()
    return {**meta, "events": events, "pool": "B_ACCUMULATION_1H"}


def events_csv(root: Path) -> str:
    store = BAccumulationStore(root)
    rows = store.list_events()
    buf = io.StringIO()
    w = csv.DictWriter(buf, fieldnames=CSV_COLS, extrasaction="ignore")
    w.writeheader()
    for r in rows:
        w.writerow({k: r.get(k) for k in CSV_COLS})
    return buf.getvalue()
'''

TEST = r'''# -*- coding: utf-8 -*-
from __future__ import annotations

import csv
import io
import tempfile
import unittest
from pathlib import Path

from b_accumulation_store import BAccumulationStore
from b_observe_api import events_csv, events_payload


class P4Api(unittest.TestCase):
    def test_list_and_csv(self):
        with tempfile.TemporaryDirectory() as td:
            root = Path(td)
            store = BAccumulationStore(root)
            store.ensure_observation_start("2026-01-01T00:00:00+00:00")
            store.upsert_event(
                market="USDⓈ-M",
                symbol="CCCUSDT",
                trigger_open_time=1700000000000,
                payload={
                    "technical_event": "pass",
                    "technical_confirmation": "pending",
                    "pending_confirm": True,
                    "event_ratio": 9.0,
                    "confirm_ratio": None,
                    "abs_return_ok": True,
                    "input_fingerprint": "abc",
                },
            )
            payload = events_payload(root)
            self.assertEqual(payload["event_count"], 1)
            self.assertEqual(payload["pool"], "B_ACCUMULATION_1H")
            csv_text = events_csv(root)
            rows = list(csv.DictReader(io.StringIO(csv_text)))
            self.assertEqual(rows[0]["symbol"], "CCCUSDT")


if __name__ == "__main__":
    unittest.main()
'''
